# HotSpring Portable Spas - Next.js App Commands
# ================================================

# STEP 1: Install dependencies
cd hotspring-spa
npm install

# STEP 2: Run in development mode (with hot reload)
npm run dev
# App will be available at: http://localhost:3000

# STEP 3: Build for production
npm run build

# STEP 4: Run in production mode (after build)
npm run start
# App will be available at: http://localhost:3000

# ================================================
# PAGES AVAILABLE:
# /                       -> Home
# /category               -> Category / Product Listing
# /product/[id]           -> Product Detail (e.g. /product/1)
# /login                  -> Login
# /register               -> Register
# /forgot-password        -> Forgot Password
# /account                -> My Account
# /account/edit           -> Edit Account
# /account/billing        -> Edit Billing Address
# /account/shipping       -> Edit Shipping Address
# /account/orders         -> Order Summary
# /account/orders/[id]    -> Order Detail (e.g. /account/orders/100000001)
# /cart                   -> Shopping Cart
# /checkout/payment       -> Payment / Checkout
# /about                  -> About Us
# /contact                -> Contact Us
# /terms                  -> Terms & Conditions
# ================================================

# REQUIREMENTS:
# Node.js v18 or higher
